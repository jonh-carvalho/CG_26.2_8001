```
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# Funções de Transformação
# ==========================================

def translate(points, translation):
    return points + np.array(translation)

def scale(points, factor):
    return points * np.array(factor)

def rotate(points, angle_degrees):
    theta = np.radians(angle_degrees)
    rot_matrix = np.array([
        [np.cos(theta), -np.sin(theta)], 
        [np.sin(theta), np.cos(theta)]
    ])
    return points @ rot_matrix.T

def reflect(points, axis='x'):
    if axis.lower() == 'x':
        return points * np.array([1, -1])
    elif axis.lower() == 'y':
        return points * np.array([-1, 1])
    return points

def shear_x(points, k):
    shear_matrix = np.array([[1, k], [0, 1]])
    return points @ shear_matrix.T


# ==========================================
# Resolução dos Exercícios
# ==========================================

# Ex 1: Translação Simples (Ponto)
p1 = np.array([2, 3])
p1_trans = translate(p1, [4, -2])

plt.figure()
plt.plot(p1[0], p1[1], marker="o", linestyle="", label="Original", color="blue")
plt.plot(p1_trans[0], p1_trans[1], marker="x", linestyle="", label="Transladado", color="red")
plt.title("Ex 1: Translação Simples")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 2 e 3: Escala Uniforme e Não Uniforme (Triângulo)
t2 = np.array([[1, 1], [3, 1], [2, 4]])
t2_uniform = scale(t2, 2)
t2_non_uniform = scale(t2, [2, 0.5])

plt.figure()
plt.plot(*zip(*t2, t2[0]), label="Original")
plt.plot(*zip(*t2_uniform, t2_uniform[0]), label="Escala Uniforme", linestyle="--")
plt.plot(*zip(*t2_non_uniform, t2_non_uniform[0]), label="Escala Não Uniforme", linestyle="-.")
plt.title("Ex 2 e 3: Escalas em um Triângulo")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 4: Rotação na Origem (Ponto)
p4 = np.array([1, 0])
p4_trans = rotate(p4, 90)

plt.figure()
plt.plot(p4[0], p4[1], marker="o", linestyle="", label="Original")
plt.plot(p4_trans[0], p4_trans[1], marker="x", linestyle="", label="Rotacionado 90°")
plt.title("Ex 4: Rotação em Torno da Origem")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 5: Rotação de um Polígono (Quadrado)
sq5 = np.array([[1, 1], [1, 4], [4, 4], [4, 1]])
sq5_trans = rotate(sq5, -45) # 45° Horário (negativo)

plt.figure()
plt.plot(*zip(*sq5, sq5[0]), label="Original")
plt.plot(*zip(*sq5_trans, sq5_trans[0]), label="Rotacionado -45°", linestyle="--")
plt.title("Ex 5: Rotação de um Polígono")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 6: Reflexão Simples (Eixo Y - Ponto)
p6 = np.array([2, 5])
p6_trans = reflect(p6, 'y')

plt.figure()
plt.plot(p6[0], p6[1], marker="o", linestyle="", label="Original")
plt.plot(p6_trans[0], p6_trans[1], marker="x", linestyle="", label="Refletido em Y")
plt.axvline(0, color='black', linewidth=0.8) # Linha do eixo Y para referência
plt.title("Ex 6: Reflexão Simples (Eixo Y)")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 7: Reflexão de um Triângulo (Eixo X)
t7 = np.array([[2, 3], [4, 3], [3, 5]])
t7_trans = reflect(t7, 'x')

plt.figure()
plt.plot(*zip(*t7, t7[0]), label="Original")
plt.plot(*zip(*t7_trans, t7_trans[0]), label="Refletido em X", linestyle="--")
plt.axhline(0, color='black', linewidth=0.8) # Linha do eixo X para referência
plt.title("Ex 7: Reflexão (Eixo X)")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 8: Cisalhamento Horizontal (k=2 - Ponto)
p8 = np.array([2, 3])
p8_trans = shear_x(p8, 2)

plt.figure()
plt.plot(p8[0], p8[1], marker="o", linestyle="", label="Original")
plt.plot(p8_trans[0], p8_trans[1], marker="x", linestyle="", label="Cisalhado")
plt.title("Ex 8: Cisalhamento Horizontal")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 9: Composição de Transformações em um Ponto
p9 = np.array([3, 2])
p9_1 = translate(p9, [1, -1])        # 1. Translação
p9_2 = rotate(p9_1, 90)              # 2. Rotação 90°
p9_3 = scale(p9_2, 2)                # 3. Escala uniforme

plt.figure()
plt.plot(p9[0], p9[1], marker="o", linestyle="", label="Original")
plt.plot(p9_3[0], p9_3[1], marker="x", linestyle="", label="Resultado Final")
plt.title("Ex 9: Composição de Transformações")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()


# Ex 10: Combinação de Transformações em uma Figura (Retângulo)
r10 = np.array([[1, 1], [5, 1], [5, 3], [1, 3]])
r10_1 = translate(r10, [-2, 3])      # 1. Translação
r10_2 = scale(r10_1, [1.5, 0.5])     # 2. Escala Não Uniforme
r10_final = reflect(r10_2, 'y')      # 3. Reflexão em Y

plt.figure()
plt.plot(*zip(*r10, r10[0]), label="Original")
plt.plot(*zip(*r10_final, r10_final[0]), label="Resultado Final", linestyle="-.")
plt.title("Ex 10: Combinação de Transformações")
plt.legend()
plt.grid(True)
plt.axis('equal')
plt.show()
```
